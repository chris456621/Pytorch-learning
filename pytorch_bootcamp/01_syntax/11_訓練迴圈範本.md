## §1 完整的訓練骨架（背下這個結構）

```python
import time, math
from pathlib import Path
import torch
import torch.nn as nn
from torch.utils.data import DataLoader


def train_one_epoch(model, loader, criterion, optimizer, device,
                    scheduler=None, scaler=None, clip=None):
    model.train()                                    # ① 切訓練模式
    total_loss, total_correct, total_n = 0.0, 0, 0

    for x, y in loader:
        x = x.to(device, non_blocking=True)          # ② 搬到 GPU
        y = y.to(device, non_blocking=True)

        optimizer.zero_grad(set_to_none=True)        # ③ 清梯度

        with torch.autocast(device_type=device.type,  # ④ 混合精度（可選）
                            dtype=torch.bfloat16,
                            enabled=scaler is not None):
            logits = model(x)                        # ⑤ forward
            loss = criterion(logits, y)              # ⑥ 算 loss

        if scaler is not None:                       # ⑦ backward
            scaler.scale(loss).backward()
            if clip:
                scaler.unscale_(optimizer)
                nn.utils.clip_grad_norm_(model.parameters(), clip)
            scaler.step(optimizer)                   # ⑧ 更新
            scaler.update()
        else:
            loss.backward()
            if clip:
                nn.utils.clip_grad_norm_(model.parameters(), clip)
            optimizer.step()

        if scheduler is not None and getattr(scheduler, "_per_batch", False):
            scheduler.step()                         # ⑨ batch 級排程

        bs = y.size(0)                               # ⑩ 累積統計（★ 用 item/detach）
        total_loss += loss.item() * bs
        total_correct += (logits.argmax(1) == y).sum().item()
        total_n += bs

    return total_loss / total_n, total_correct / total_n


@torch.no_grad()                                     # ★ 裝飾器版本，最乾淨
def evaluate(model, loader, criterion, device):
    model.eval()                                     # ★ 一定要
    total_loss, total_correct, total_n = 0.0, 0, 0
    for x, y in loader:
        x = x.to(device, non_blocking=True)
        y = y.to(device, non_blocking=True)
        logits = model(x)
        loss = criterion(logits, y)
        bs = y.size(0)
        total_loss += loss.item() * bs
        total_correct += (logits.argmax(1) == y).sum().item()
        total_n += bs
    return total_loss / total_n, total_correct / total_n
```

### 十個步驟，一個都不能漏

| # | 步驟 | 漏掉的症狀 |
|---|---|---|
| ① | `model.train()` | 第二個 epoch 起 dropout/BN 行為錯誤 |
| ② | `.to(device)` | `RuntimeError: expected all tensors on same device` |
| ③ | `zero_grad()` | 梯度累加 → loss 爆炸 |
| ⑤ | `model(x)` 而非 `model.forward(x)` | hook 失效 |
| ⑥ | loss 的 shape 對齊 | 靜默廣播 → loss 完全錯 |
| ⑦ | `loss.backward()` | 梯度全是 None，loss 不動 |
| ⑧ | `optimizer.step()` | 參數永遠不更新 |
| ⑨ | scheduler 呼叫位置 | lr 不變或降得太快 |
| ⑩ | `loss.item()` 而非 `loss` | 記憶體累積到 OOM |
| — | 驗證時 `model.eval()` + `no_grad()` | 準確率偏低、記憶體爆 |

                "config": asdict(cfg),

        if stopper.should_stop:
            print(f"Early stop at epoch {ep}")
            break

**AverageMeter（工程界標準做法）：**

```python
class AverageMeter:
    def __init__(self): self.reset()
    def reset(self): self.sum = 0.0; self.count = 0
    def update(self, val, n=1): self.sum += val * n; self.count += n
    @property
    def avg(self): return self.sum / max(self.count, 1)

loss_meter = AverageMeter()
loss_meter.update(loss.item(), y.size(0))
print(loss_meter.avg)
```

## §4 多指標評估（分類任務不能只看 accuracy）★

```python
@torch.no_grad()
def evaluate_full(model, loader, device, num_classes):
    model.eval()
    all_logits, all_y = [], []
    for x, y in loader:
        all_logits.append(model(x.to(device)).cpu())
        all_y.append(y)
    logits = torch.cat(all_logits); y = torch.cat(all_y)
    pred = logits.argmax(1)

    acc = (pred == y).float().mean().item()
    cm = torch.zeros(num_classes, num_classes, dtype=torch.long)
    for t, p in zip(y, pred):
        cm[t, p] += 1

    tp = cm.diag().float()
    precision = tp / cm.sum(0).clamp(min=1)          # 每類 precision
    recall    = tp / cm.sum(1).clamp(min=1)          # 每類 recall
    f1 = 2 * precision * recall / (precision + recall).clamp(min=1e-12)

    return {
        "acc": acc,
        "macro_f1": f1.mean().item(),                # ★ 不平衡資料要看這個
        "per_class_f1": f1.tolist(),
        "confusion": cm,
    }
```

> 🔥 **類別不平衡時 accuracy 會騙人。**
> 例如 90% 資料是「不漲」，模型全部猜「不漲」就有 90% accuracy，但完全沒用。
> **一定要看 macro-F1 和混淆矩陣。**

## §8 多 seed 實驗（★ 做研究的基本要求）

```python
results = []
for seed in [0, 1, 2, 3, 4]:
    set_seed(seed)
    model = MyModel().to(device)
    acc = run_training(model, ...)
    results.append(acc)

import numpy as np
print(f"{np.mean(results):.4f} ± {np.std(results):.4f}  (n=5)")
```

> 🔥 **只跑一個 seed 就說「A 比 B 好」是無效的結論。**
> 深度學習的 seed 之間差異常常有 ±1%，而很多論文宣稱的改進也只有 0.5%。
> 報告 mean ± std 是最基本的科學誠實。
